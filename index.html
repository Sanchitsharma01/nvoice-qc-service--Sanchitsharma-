    <meta charset="UTF-8">

    <title>Invoice QC Console</title>

    <script src="https://cdn.tailwindcss.com"></script>

    <script src="https://unpkg.com/vue@3/dist/vue.global.js"></script>

</head>

<body class="bg-gray-50 text-gray-800 font-sans p-8">

    <div id="app" class="max-w-5xl mx-auto">

        <header class="mb-10 text-center">

            <h1 class="text-3xl font-bold text-indigo-700">Invoice QC Console</h1>

            <p class="text-gray-500">Automated Invoice Extraction & Validation</p>

        </header>


        <div class="bg-white p-8 rounded shadow-md text-center border-dashed border-2 border-gray-300 mb-8">

            <input type="file" ref="fileInput" multiple @change="handleFileSelect" class="hidden" id="upload">

            <label for="upload" class="cursor-pointer text-indigo-600 font-bold hover:text-indigo-800">

                Click to Upload PDF Invoices

            </label>

            <p v-if="files.length" class="mt-2 text-sm text-gray-600">{{ files.length }} files selected</p>

            <button @click="runAnalysis" :disabled="loading || !files.length" class="mt-4 bg-indigo-600 text-white px-6 py-2 rounded disabled:opacity-50">

                {{ loading ? 'Processing...' : 'Run Analysis' }}

            </button>

        </div>


        <div v-if="report" class="grid grid-cols-3 gap-4 mb-8">

            <div class="bg-white p-5 rounded shadow border-l-4 border-blue-500">

                <p class="text-gray-500 text-sm">Processed</p>

                <p class="text-2xl font-bold">{{ report.summary.total_processed }}</p>

            </div>

            <div class="bg-white p-5 rounded shadow border-l-4 border-green-500">

                <p class="text-gray-500 text-sm">Valid</p>

                <p class="text-2xl font-bold text-green-600">{{ report.summary.valid_invoices }}</p>

            </div>

            <div class="bg-white p-5 rounded shadow border-l-4 border-red-500">

                <p class="text-gray-500 text-sm">Invalid</p>

                <p class="text-2xl font-bold text-red-600">{{ report.summary.invalid_invoices }}</p>

            </div>

        </div>


        <div v-if="report" class="space-y-4">

            <div v-for="item in report.details" :key="item.invoice_id" class="bg-white p-4 rounded shadow">

                <div class="flex justify-between items-center">

                    <div>

                        <span :class="item.is_valid ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'" class="px-2 py-1 rounded text-xs font-bold">

                            {{ item.is_valid ? 'VALID' : 'INVALID' }}

                        </span>

                        <span class="ml-3 font-semibold">{{ item.invoice_id }}</span>

                    </div>

                </div>

                <div v-if="!item.is_valid" class="mt-3 bg-red-50 p-2 rounded text-sm text-red-700">

                    <ul class="list-disc list-inside">

                        <li v-for="err in item.errors">{{ err }}</li>

                    </ul>

                </div>

            </div>

        </div>

    </div>

    <script>

        const { createApp } = Vue

        createApp({

            data() { return { files: [], loading: false, report: null } },

            methods: {

                handleFileSelect(e) { this.files = Array.from(e.target.files); this.report = null; },

                async runAnalysis() {

                    this.loading = true;

                    const fd = new FormData();

                    this.files.forEach(f => fd.append('files', f));

                    try {

                        const res = await fetch('http://127.0.0.1:8000/extract-and-validate-pdfs', { method: 'POST', body: fd });

                        const data = await res.json();

                        this.report = data.validation_report;

                    } catch (e) { alert("API Error. Is backend running?"); }

                    finally { this.loading = false; }

                }

            }

        }).mount('#app')

    </script>

</body>

</html>
